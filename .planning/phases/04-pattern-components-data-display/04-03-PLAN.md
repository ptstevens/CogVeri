3. Create `frontend/src/components/ui/patterns/Charts/BarChart.tsx`:
   ```typescript
   import {
     BarChart as RechartsBarChart,
     Bar,
     XAxis,
     YAxis,
     CartesianGrid,
     Tooltip,
     Legend,
     ResponsiveContainer,
   } from 'recharts';

   export interface BarChartProps {
     data: ChartDataPoint[];
     height?: number;
     dataKey?: string;
     /** Show secondary data series */
     secondaryDataKey?: string;
     /** Show legend (default: false) */
     showLegend?: boolean;
     /** Show grid (default: true) */
     showGrid?: boolean;
     className?: string;
   }

   export function BarChart({
     data,
     height = 300,
     dataKey = 'value',
     secondaryDataKey,
     showLegend = false,
     showGrid = true,
     className,
   }: BarChartProps) {
     return (
       <div className={className} role="img" aria-label="Bar chart">
         <ResponsiveContainer width="100%" height={height}>
           <RechartsBarChart data={data}>
             {showGrid && <CartesianGrid strokeDasharray="3 3" stroke={CHART_COLORS.grayLight} />}
             <XAxis dataKey="name" tick={{ fill: CHART_COLORS.gray, fontSize: 12 }} />
             <YAxis tick={{ fill: CHART_COLORS.gray, fontSize: 12 }} />
             <Tooltip
               contentStyle={{
                 backgroundColor: 'white',
                 border: '1px solid ' + CHART_COLORS.grayLight,
                 borderRadius: '8px',
                 fontSize: '14px',
               }}
             />
             {showLegend && <Legend />}
             <Bar dataKey={dataKey} fill={CHART_COLORS.primary} radius={[4, 4, 0, 0]} />
             {secondaryDataKey && (
               <Bar dataKey={secondaryDataKey} fill={CHART_COLORS.accent} radius={[4, 4, 0, 0]} />
             )}
           </RechartsBarChart>
         </ResponsiveContainer>
       </div>
     );
   }
   ```

5. Create `frontend/src/components/ui/patterns/Charts/PieChart.tsx`:
   ```typescript
   import { PieChart as RechartsPieChart, Pie, Cell, Tooltip, Legend, ResponsiveContainer } from 'recharts';

   export interface PieChartProps {
     data: ChartDataPoint[];
     height?: number;
     /** Show legend (default: true for pie charts) */
     showLegend?: boolean;
     /** Inner radius for donut chart (default: 0 = full pie) */
     innerRadius?: number;
     className?: string;
   }

   const PIE_COLORS = [
     CHART_COLORS.primary,
     CHART_COLORS.accent,
     CHART_COLORS.primaryLight,
     CHART_COLORS.accentLight,
   ];

   export function PieChart({ data, height = 300, showLegend = true, innerRadius = 0, className }: PieChartProps) {
     return (
       <div className={className} role="img" aria-label="Pie chart">
         <ResponsiveContainer width="100%" height={height}>
           <RechartsPieChart>
             <Pie
               data={data}
               cx="50%"
               cy="50%"
               innerRadius={innerRadius}
               outerRadius={80}
               dataKey="value"
               label={({ name, percent }) => `${name} ${(percent * 100).toFixed(0)}%`}
               labelLine={false}
             >
               {data.map((_, index) => (
                 <Cell key={`cell-${index}`} fill={PIE_COLORS[index % PIE_COLORS.length]} />
               ))}
             </Pie>
             <Tooltip />
             {showLegend && <Legend />}
           </RechartsPieChart>
         </ResponsiveContainer>

         {/* Screen reader fallback table */}
         <table className="sr-only">
           <caption>Chart Data</caption>
           <thead><tr><th>Name</th><th>Value</th></tr></thead>
           <tbody>
             {data.map((item) => (
               <tr key={item.name}><td>{item.name}</td><td>{item.value}</td></tr>
             ))}
           </tbody>
         </table>
       </div>
     );
   }
   ```
